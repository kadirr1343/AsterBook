cmake_minimum_required(VERSION 3.25)
project(AsterBook VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>
int main(){std::expected<int,int> x=1;return *x-1;}" HAS_EXPECTED)
if(NOT HAS_EXPECTED)
 message(FATAL_ERROR "C++23 std::expected is required; use GCC 13+")
endif()
option(ASTERBOOK_SANITIZE "ASan and UBSan" OFF)
add_library(asterbook_core src/reference.cpp src/price_level.cpp src/io.cpp benchmarks/harness.cpp)
target_include_directories(asterbook_core PUBLIC include)
target_compile_options(asterbook_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(ASTERBOOK_SANITIZE)
 target_compile_options(asterbook_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(asterbook_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(asterbook apps/main.cpp)
target_link_libraries(asterbook PRIVATE asterbook_core)
add_executable(asterbook_tests tests/test.cpp)
target_link_libraries(asterbook_tests PRIVATE asterbook_core)
enable_testing()
add_test(NAME correctness COMMAND asterbook_tests)
add_test(NAME demo COMMAND asterbook demo)
add_test(NAME replay_verify COMMAND asterbook verify --input ${CMAKE_CURRENT_SOURCE_DIR}/examples/sweep.csv)

add_executable(allocation_tests tests/allocation.cpp)
target_link_libraries(allocation_tests PRIVATE asterbook_core)
add_test(NAME allocation_fail_stop COMMAND allocation_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:asterbook> ${CMAKE_CURRENT_SOURCE_DIR}/examples/sweep.csv)
add_test(NAME report_validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/report.py ${CMAKE_CURRENT_SOURCE_DIR}/tools/report.py)
